=== docker-safe compose config (profile observability) ===
rc=0
--- prometheus block ---
  prometheus:
    profiles:
      - observability
    container_name: alphard-prometheus
    entrypoint:
      - sh
      - -c
      - |
        set -e
        mkdir -p /etc/prometheus /prometheus
        echo "$$PROM_YML_B64" | base64 -d > /etc/prometheus/prometheus.yml
        # BUGFIX (issue #209): /prometheus is a named volume — Docker creates
        # it root-owned, but prom/prometheus runs as in-image `nobody` (uid 65534).
        # Without chown, queries.active write fails with EPERM and container
        # restart-loops. Same userns-mapping trap as bind-mounts, different fix.
        chown -R nobody:nogroup /prometheus
        exec /bin/prometheus \
          --config.file=/etc/prometheus/prometheus.yml \
          --storage.tsdb.path=/prometheus \
          --storage.tsdb.retention.time=30d \
          --web.listen-address=0.0.0.0:9090 \
          --web.enable-lifecycle
    environment:
      PROM_YML_B64: ""
    healthcheck:
      test:
        - CMD-SHELL
        - wget -q -O /dev/null http://127.0.0.1:9090/-/healthy
      timeout: 10s
      interval: 30s
      retries: 3
      start_period: 1m0s
    image: prom/prometheus:latest
    networks:
      alphard-net:
        aliases:
          - prometheus
          - alphard-prometheus
    ports:
      - mode: ingress
        target: 9090
        published: "9090"
        protocol: tcp
    restart: unless-stopped
    security_opt:
      - apparmor=unconfined
    volumes:
      - type: volume
        source: alphard-prometheus-data
        target: /prometheus
        volume: {}
